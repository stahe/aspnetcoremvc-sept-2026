    8. Authentification et autorisation

    Deux questions distinctes se posent pour chaque requête :

    • qui est l'utilisateur ? C'est l'authentification : il s'est identifié par un login et un mot de passe, et chaque requête suivante doit prouver qu'elle vient de lui ;
    • a-t-il le droit d'accéder à cette page ? C'est l'autorisation : dans l'étude de cas, seul un utilisateur de rôle ADMIN gère les médecins et les clients.

    Le serveur étant sans état (chapitre 6), la preuve d'identité doit voyager avec chaque requête. On utilise un jeton JWT ([JSON Web Token]) : après la connexion, le serveur fabrique un jeton qui contient l'identité de l'utilisateur et le signe avec une clé secrète. Le jeton est déposé dans un cookie ; le navigateur le renvoie à chaque requête ; le serveur vérifie la signature. L'utilisateur peut lire son jeton, mais pas le modifier : toute modification rend la signature invalide.

    [ASP.NET Core] sépare nettement les deux questions :

    • l'authentification est l'affaire d'un middleware ([UseAuthentication]) et d'un schéma d'authentification : ici le schéma [JwtBearer], fourni par le paquet [NuGet] [Microsoft.AspNetCore.Authentication.JwtBearer]. Pour chaque requête, il cherche un jeton, le vérifie et en tire l'utilisateur, [HttpContext.User] ;
    • l'autorisation est l'affaire de l'attribut [[Authorize]], vérifié par le middleware [UseAuthorization] : il compare ce qu'exige l'action à ce qu'on sait de l'utilisateur.

    Les exemples de ce chapitre utilisent deux paquets [NuGet] : [Microsoft.AspNetCore.Authentication.JwtBearer] et [BCrypt.Net-Next] (le hachage des mots de passe). Ils sont déclarés dans le fichier projet ; la commande [dotnet build] (ou [dotnet run]) les télécharge depuis nuget.org lors de la première compilation : il faut donc une connexion Internet.

    namespace Exemple25.Auth;

public record Utilisateur(string Login, string Nom, string Role, string HachageMotDePasse);

// les utilisateurs, en mémoire (le chapitre suivant les mettra dans une base)
public class UtilisateursService
{
    private readonly List<Utilisateur> utilisateurs =
    [
        // on ne conserve JAMAIS un mot de passe, seulement son hachage
        Creer("marie", "Marie Pélissier", "USER", "marie"),
        Creer("admin", "Administrateur", "ADMIN", "admin"),
    ];

    public Utilisateur? Chercher(string login) =>
        utilisateurs.Find(u => u.Login == login);

    // bcrypt, avec un coût de 10 : 2^10 tours de calcul, volontairement lents
    private static Utilisateur Creer(string login, string nom, string role, string motDePasse) =>
        new(login, nom, role, BCrypt.Net.BCrypt.HashPassword(motDePasse, 10));
}
    • ligne 3 : un utilisateur, avec son rôle ;
    • lignes 10-12 : on ne conserve jamais un mot de passe, seulement son hachage : une empreinte calculée par une fonction à sens unique. On peut vérifier qu'un mot de passe correspond à une empreinte, mais pas retrouver le mot de passe à partir de l'empreinte ;
    • lignes 19-20 : [BCrypt.Net.BCrypt.HashPassword(motDePasse, 10)] calcule le hachage avec l'algorithme [bcrypt]. Le 10 est le « coût » : [bcrypt] est volontairement lent (2¹⁰ tours de calcul), pour décourager qui voudrait essayer des millions de mots de passe. Deux hachages du même mot de passe sont différents (un « sel » aléatoire est ajouté) : c'est [BCrypt.Verify] qui sait les vérifier.

    using System.Text;
using Microsoft.IdentityModel.JsonWebTokens;
using Microsoft.IdentityModel.Tokens;

namespace Exemple25.Auth;

public class AuthService(UtilisateursService utilisateurs, IConfiguration configuration)
{
    // le nom du cookie qui transporte le jeton
    public const string Cookie = "jeton";

    // la clé qui signe les jetons, lue dans appsettings.json (section Jwt)
    public static SymmetricSecurityKey Cle(IConfiguration configuration) =>
        new(Encoding.UTF8.GetBytes(configuration["Jwt:Secret"]!));

    // null si le login est inconnu ou le mot de passe faux
    public Utilisateur? Valider(string? login, string? motDePasse)
    {
        var utilisateur = utilisateurs.Chercher(login ?? "");
        if (utilisateur is null || motDePasse is null)
        {
            return null;
        }
        return BCrypt.Net.BCrypt.Verify(motDePasse, utilisateur.HachageMotDePasse) ? utilisateur : null;
    }

    // le jeton JWT de l'utilisateur, signé avec la clé secrète
    public string CreerJeton(Utilisateur utilisateur)
    {
        var descripteur = new SecurityTokenDescriptor
        {
            // le contenu du jeton : les "claims" (affirmations sur l'utilisateur)
            Claims = new Dictionary<string, object>
            {
                ["sub"] = utilisateur.Login,
                ["nom"] = utilisateur.Nom,
                ["role"] = utilisateur.Role,
            },
            Expires = DateTime.UtcNow.AddMinutes(configuration.GetValue<int>("Jwt:DureeMinutes")),
            SigningCredentials = new SigningCredentials(Cle(configuration), SecurityAlgorithms.HmacSha256),
        };
        return new JsonWebTokenHandler().CreateToken(descripteur);
    }
}

    using System.Text.Encodings.Web;
using System.Text.Unicode;
using Exemple25.Auth;
using Microsoft.AspNetCore.Authentication.JwtBearer;
using Microsoft.Extensions.FileProviders;
using Microsoft.IdentityModel.Tokens;

var builder = WebApplication.CreateBuilder(args);
builder.Services.AddControllersWithViews();
builder.Services.AddSingleton(HtmlEncoder.Create(UnicodeRanges.All));
builder.Services.AddSingleton<UtilisateursService>();
builder.Services.AddSingleton<AuthService>();

// l'authentification : un jeton JWT, vérifié à chaque requête
builder.Services
    .AddAuthentication(JwtBearerDefaults.AuthenticationScheme)
    .AddJwtBearer(options =>
    {
        options.MapInboundClaims = false; // garder les noms des claims tels quels ("sub", "role"...)
        options.TokenValidationParameters = new TokenValidationParameters
        {
            ValidateIssuerSigningKey = true,
            IssuerSigningKey = AuthService.Cle(builder.Configuration), // la signature doit être la nôtre
            ValidateLifetime = true,                                   // le jeton ne doit pas être expiré
            ValidateIssuer = false,
            ValidateAudience = false,
            NameClaimType = "nom",  // User.Identity.Name
            RoleClaimType = "role", // [Authorize(Roles = ...)], User.IsInRole(...)
        };
        options.Events = new JwtBearerEvents
        {
            // 1. le jeton est lu dans le cookie (et non dans l'entête Authorization)
            OnMessageReceived = contexte =>
            {
                contexte.Token = contexte.Request.Cookies[AuthService.Cookie];
                return Task.CompletedTask;
            },
            // 2. page protégée sans jeton valide : direction la page de connexion
            OnChallenge = contexte =>
            {
                contexte.HandleResponse(); // on fabrique nous-mêmes la réponse (pas de 401)
                contexte.Response.Redirect("/login");
                return Task.CompletedTask;
            },
        };
    });
builder.Services.AddAuthorization();

var app = builder.Build();

app.UseStaticFiles(new StaticFileOptions
{
    FileProvider = new PhysicalFileProvider(Path.GetFullPath(Path.Combine(builder.Environment.ContentRootPath, "..", "commun", "bootstrap"))),
    RequestPath = "/bootstrap",
});
app.UseAuthentication(); // qui est-ce ? -> HttpContext.User
app.UseAuthorization();  // a-t-il le droit ? -> [Authorize]

app.MapControllers();
app.Run();

    Commentons ce code :

    • lignes 16-17 : on déclare le schéma d'authentification [JwtBearer], et on en fait le schéma par défaut ;
    • ligne 19 : par défaut, [JwtBearer] renomme certains claims avec des noms longs hérités des technologies Microsoft ([role] deviendrait [http://schemas.microsoft.com/ws/2008/06/identity/claims/role]). On garde les noms du jeton ;
    • lignes 20-26 : les vérifications faites sur chaque jeton : la signature (avec notre clé) et la date d'expiration. Nos jetons n'indiquent ni émetteur ([issuer]) ni destinataire ([audience]) : on ne les vérifie pas ;
    • lignes 27-28 : quel claim contient le nom de l'utilisateur ([User.Identity.Name]), et lequel contient son rôle ([[Authorize(Roles = ...)]], [User.IsInRole(...)]) ;
    • lignes 33-36 : les événements de [JwtBearer] permettent d'adapter son comportement. Par défaut, il cherche le jeton dans l'entête [Authorization: Bearer ...] de la requête : c'est ainsi que procède un client [JavaScript] ou une application mobile. Un navigateur n'ajoute pas cet entête de lui-même : on va chercher le jeton dans le cookie ;
    • lignes 39-42 : le challenge a lieu quand une page protégée est demandée sans jeton valide. Par défaut, la réponse est une erreur 401, sans contenu : utile à un client [JavaScript], déroutante pour un utilisateur. On redirige vers la page de connexion. [HandleResponse()] indique à [JwtBearer] que la réponse est prise en charge ;
    • lignes 56-57 : les deux middlewares, dans cet ordre : d'abord établir qui est l'utilisateur, ensuite vérifier ses droits.

    using Exemple25.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Exemple25.Controllers;

public class AuthController(AuthService auth, IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Accueil() => Redirect("/profil");

    [HttpGet("login")]
    public IActionResult Formulaire() => View("Login");

    [HttpPost("login")]
    public IActionResult Connexion(string? login, string? motDePasse)
    {
        var utilisateur = auth.Valider(login, motDePasse);
        if (utilisateur is null)
        {
            // un message volontairement vague : on ne dit pas si c'est le login ou le mot de passe
            ViewData["Erreur"] = "Login ou mot de passe incorrect.";
            ViewData["Login"] = login;
            return View("Login");
        }
        // le jeton, dans un cookie que le JavaScript de la page ne peut pas lire
        Response.Cookies.Append(AuthService.Cookie, auth.CreerJeton(utilisateur), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DureeMinutes")),
        });
        return Redirect("/profil");
    }

    // la déconnexion modifie l'état de l'utilisateur : c'est un POST
    [HttpPost("logout")]
    public IActionResult Deconnexion()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }
}

    Commentons ce code :

    • lignes 14-23 : la connexion. En cas d'échec, le formulaire est réaffiché avec un message volontairement vague : dire « ce login n'existe pas » aiderait un attaquant à trouver des logins valides ;
    • lignes 26-32 : en cas de succès, le jeton est déposé dans un cookie [HttpOnly] : le [JavaScript] de la page ne peut pas le lire, ce qui protège le jeton si un script malveillant parvenait à s'exécuter dans la page. [SameSite = Lax] : le cookie n'accompagne pas les requêtes POST émises depuis un autre site, ce qui protège contre la falsification de requêtes ([CSRF]). La durée de vie du cookie est celle du jeton ;
    • lignes 36-40 : la déconnexion efface le cookie ([Cookies.Delete]). C'est un POST : elle modifie l'état de l'utilisateur (cf. exemple 21).

    using System.Collections.Concurrent;

namespace Exemple27.Auth;

// un captcha : son identifiant, et son image (SVG)
public record Captcha(string Id, string Svg);

// singleton : les captchas en attente sont gardés en mémoire par le serveur
public class CaptchaService
{
    private static readonly TimeSpan DureeVie = TimeSpan.FromMinutes(10);

    // identifiant -> (réponse attendue, date d'expiration).
    // ConcurrentDictionary : un dictionnaire utilisable par plusieurs requêtes simultanées
    private readonly ConcurrentDictionary<string, (string Texte, DateTime Expire)> enAttente = new();

    public Captcha Generer()
    {
        var texte = DessinCaptcha.TexteAleatoire(5);
        var id = Guid.NewGuid().ToString(); // un identifiant aléatoire, impossible à deviner
        enAttente[id] = (texte, DateTime.UtcNow + DureeVie);
        return new Captcha(id, DessinCaptcha.Svg(texte));
    }

    // le captcha est CONSOMMÉ dans tous les cas : une image ne sert qu'une fois
    public bool Verifier(string? id, string? saisie)
    {
        if (id is null || saisie is null || !enAttente.TryRemove(id, out var attendu))
        {
            return false;
        }
        return attendu.Expire > DateTime.UtcNow
            && string.Equals(attendu.Texte, saisie.Trim(), StringComparison.OrdinalIgnoreCase);
    }
}

    Commentons ce code :

    • ligne 15 : les réponses attendues sont gardées par le serveur, sous un identifiant aléatoire. La page ne contient que l'image et l'identifiant : la réponse ne quitte jamais le serveur. Le service est un singleton, utilisé par des requêtes simultanées : [ConcurrentDictionary] est un dictionnaire prévu pour cela ;
    • lignes 17-22 : un texte de cinq caractères tiré au hasard, mémorisé, et dessiné dans une image. Le dessin est confié à la classe [DessinCaptcha] (fichier [Auth/DessinCaptcha.cs], repris de l'étude de cas et commenté ligne à ligne) : chaque caractère est un ensemble de traits, agrandi, penché, déformé au hasard, et l'image est brouillée par trois courbes. Les caractères faciles à confondre (0 et O, 1, I et L) sont exclus ;

    namespace Exemple27.Auth;

// singleton : compte les échecs de connexion de chaque adresse IP
public class LimiteurService
{
    private const int MaxEchecs = 3;
    private static readonly TimeSpan DureeBlocage = TimeSpan.FromMinutes(1);

    private class Compteur
    {
        public int Echecs;
        public DateTime BloqueJusqua;
    }

    private readonly Dictionary<string, Compteur> compteurs = [];
    private readonly Lock verrou = new(); // plusieurs requêtes simultanées : un seul accès à la fois

    // le nombre de secondes de blocage restantes (0 : pas bloquée)
    public int SecondesBlocage(string ip)
    {
        lock (verrou)
        {
            if (!compteurs.TryGetValue(ip, out var c)) return 0;
            var reste = c.BloqueJusqua - DateTime.UtcNow;
            return reste > TimeSpan.Zero ? (int)Math.Ceiling(reste.TotalSeconds) : 0;
        }
    }

    public void Echec(string ip)
    {
        lock (verrou)
        {
            if (!compteurs.TryGetValue(ip, out var c))
            {
                c = new Compteur();
                compteurs[ip] = c;
            }
            c.Echecs++;
            if (c.Echecs >= MaxEchecs)
            {
                c.BloqueJusqua = DateTime.UtcNow + DureeBlocage;
                c.Echecs = 0; // après le blocage, on repart de zéro
            }
        }
    }

    public void Reussite(string ip)
    {
        lock (verrou)
        {
            compteurs.Remove(ip);
        }
    }
}
    • lignes 6-7 : trois échecs bloquent l'adresse IP une minute ;
    • ligne 16 : le dictionnaire des compteurs est partagé par toutes les requêtes. L'instruction [lock (verrou) { ... }] garantit qu'une seule requête à la fois exécute le bloc : deux échecs simultanés ne peuvent pas se perdre (cf. [Interlocked], chapitre 6). [Lock] est le type de verrou de [.NET] 9 et suivants ;
    • lignes 19-25 : le temps de blocage restant ;
    • lignes 29-42 : un échec est compté ; au troisième, l'adresse est bloquée ;
    • lignes 47-51 : une réussite remet le compteur à zéro.

    using Exemple27.Auth;
using Microsoft.AspNetCore.Mvc;

namespace Exemple27.Controllers;

public class AuthController(AuthService auth, CaptchaService captchas, LimiteurService limiteur,
    IConfiguration configuration) : Controller
{
    [HttpGet("/")]
    public IActionResult Accueil() => Redirect("/profil");

    [HttpGet("login")]
    public IActionResult Formulaire() => Login(null, null);

    [HttpPost("login")]
    public IActionResult Connexion(string? login, string? motDePasse, string? captchaId, string? captcha)
    {
        // l'adresse IP du client
        var ip = HttpContext.Connection.RemoteIpAddress?.ToString() ?? "?";

        // 1. une adresse bloquée n'est même pas écoutée
        var secondes = limiteur.SecondesBlocage(ip);
        if (secondes > 0)
        {
            return Login(login, $"Trop de tentatives : réessayez dans {secondes} s.");
        }
        // 2. le captcha d'abord (peu coûteux), le mot de passe (bcrypt, coûteux) ensuite
        if (!captchas.Verifier(captchaId, captcha))
        {
            limiteur.Echec(ip);
            return Login(login, "Le code de l'image est incorrect.");
        }
        var utilisateur = auth.Valider(login, motDePasse);
        if (utilisateur is null)
        {
            limiteur.Echec(ip);
            return Login(login, "Login ou mot de passe incorrect.");
        }
        limiteur.Reussite(ip);
        Response.Cookies.Append(AuthService.Cookie, auth.CreerJeton(utilisateur), new CookieOptions
        {
            HttpOnly = true,
            SameSite = SameSiteMode.Lax,
            MaxAge = TimeSpan.FromMinutes(configuration.GetValue<int>("Jwt:DureeMinutes")),
        });
        return Redirect("/profil");
    }

    [HttpPost("logout")]
    public IActionResult Deconnexion()
    {
        Response.Cookies.Delete(AuthService.Cookie);
        return Redirect("/login");
    }

    // chaque affichage du formulaire fabrique un NOUVEAU captcha
    private ViewResult Login(string? login, string? erreur)
    {
        ViewData["Login"] = login;
        ViewData["Erreur"] = erreur;
        return View("Login", captchas.Generer());
    }
}
